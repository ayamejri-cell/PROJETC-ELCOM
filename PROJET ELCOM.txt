def get_ip_address():

    """Get the Raspberry Pi's IP address - SIMPLE VERSION"""

    try:

        # Method 1: Connect to external server to get our IP

        s = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)

        s.connect(('8.8.8.8', 80))

        ip = s.getsockname()[0]

        s.close()

        return ip

    except:

        try:

            # Method 2: Get from hostname

            return socket.gethostbyname(socket.gethostname())

        except:

            return '127.0.0.1'



def get_all_ips_simple():

    """Get all IPs - simple version without netifaces"""

    ips = {

        'hostname': socket.gethostname(),

        'primary': get_ip_address()

    }

    

    # Try to get more info from ip command

    try:

        import subprocess

        result = subprocess.run(['ip', 'addr', 'show'], 

                              capture_output=True, text=True, timeout=2)

        

        current_iface = None

        for line in result.stdout.split('\n'):

            line = line.strip()

            if ':' in line and line[0].isdigit():

                # Interface line like "2: eth0: ..."

                parts = line.split(':')

                if len(parts) > 1:

                    current_iface = parts[1].strip()

                    ips[current_iface] = []

            elif 'inet ' in line and current_iface:

                # IP address line

                parts = line.split()

                if len(parts) > 1:

                    ip = parts[1].split('/')[0]

                    if ip != '127.0.0.1':

                        ips[current_iface].append(ip)

    except:

        pass

    

    return ips

def validate_position_advanced(x, y, z, r):

    """Validate position"""

    if not (WORKSPACE_LIMITS['x'][0] <= x <= WORKSPACE_LIMITS['x'][1]):

        return False, f"X={x:.1f} out of range"

    if not (WORKSPACE_LIMITS['y'][0] <= y <= WORKSPACE_LIMITS['y'][1]):

        return False, f"Y={y:.1f} out of range"

    if not (WORKSPACE_LIMITS['z'][0] <= z <= WORKSPACE_LIMITS['z'][1]):

        return False, f"Z={z:.1f} out of range"

    if not (WORKSPACE_LIMITS['r'][0] <= r <= WORKSPACE_LIMITS['r'][1]):

        return False, f"R={r:.1f} out of range"

    

    radial_distance = math.sqrt(x**2 + y**2)

    if radial_distance < WORKSPACE_LIMITS['min_radius']:

        return False, f"Too close to base"

    if radial_distance > WORKSPACE_LIMITS['max_radius']:

        return False, f"Out of reach"

    

    return True, None



def move_with_timeout(dobot, x, y, z, r, timeout=15, retries=2):  # Reduced timeout and retries for speed

    """Ultra-fast movement with aggressive timeout"""

    for attempt in range(retries + 1):

        result_queue = Queue()

        

        def move_thread():

            try:

                dobot.move_to(x, y, z, r, wait=True)

                result_queue.put(('success', None))

            except Exception as e:

                result_queue.put(('error', str(e)))

        

        thread = threading.Thread(target=move_thread)

        thread.daemon = True

        thread.start()

        

        try:

            result, error = result_queue.get(timeout=timeout)

            if result == 'success':

                return True, None

            else:

                if attempt < retries:

                    time.sleep(0.1)  # Minimal retry delay

                    continue

                return False, error

        except Empty:

            if attempt < retries:

                time.sleep(0.1)  # Minimal retry delay

                continue

            return False, f"Timeout after {timeout}s - MAXIMUM SPEED MODE"

    

    return False, "Max retries exceeded in ultra-fast mode"

        pose = dobot.pose()

                # Calculate differences

                diff_x = abs(new_x - x)

def execute_trajectory_async(trajectory):

    """Execute trajectory with PRECISE movement validation"""

    global is_executing, stop_flag

    is_executing = True

    stop_flag = False

    

    try:

        # Set MAXIMUM speed for trajectory execution

        original_velocity = getattr(dobot, '_velocity', DEFAULT_VELOCITY)

        original_acceleration = getattr(dobot, '_acceleration', DEFAULT_ACCELERATION)

        

        # Push to absolute maximum safe speeds

        try:

            dobot.speed(velocity=FAST_TRAJECTORY_VELOCITY, acceleration=FAST_TRAJECTORY_ACCELERATION)

            print(f"Ã°Å¸â€Â¥ ULTRA-HIGH SPEED MODE: {FAST_TRAJECTORY_VELOCITY}/{FAST_TRAJECTORY_ACCELERATION}")

        except:

            # Fallback to maximum available speed

            try:

                dobot.speed(velocity=255, acceleration=255)  # Hardware maximum

                print(f"Ã°Å¸â€Â¥ MAXIMUM HARDWARE SPEED: 255/255")

            except:

                pass

        

        print(f"Ã°Å¸Å½Â¯ Executing {len(trajectory)} steps with PRECISE positioning")

        

        execution_log = []

        

        for i, step in enumerate(trajectory):

            if stop_flag:

                execution_log.append(f"Ã°Å¸â€ºâ€˜ Execution stopped at step {i+1}")

                break

            

            step_log = f"Step {i+1}: "

            

            # Validate step structure

            if not isinstance(step, dict):

                step_log += f"Ã¢ÂÅ’ Invalid step format: {type(step)}"

                print(step_log)

                execution_log.append(step_log)

                continue

            

            # Handle gripper commands FIRST (precise case handling)

            if 'gripper' in step:

                gripper_cmd = step['gripper']

                # Handle various case formats

                if isinstance(gripper_cmd, str):

                    gripper_lower = gripper_cmd.lower().strip()

                    if gripper_lower in ['close', 'closed', '1', 'true']:

                        dobot.grip(True)

                        step_log += "Ã¢Å“â€¹ Gripper CLOSED "

                    elif gripper_lower in ['open', 'opened', '0', 'false']:

                        dobot.grip(False)

                        step_log += "Ã°Å¸â€˜Â Gripper OPENED "

                    else:

                        step_log += f"Ã¢Ââ€œ Unknown gripper command: '{gripper_cmd}' "

                else:

                    step_log += f"Ã¢ÂÅ’ Invalid gripper command type: {type(gripper_cmd)} "

                

                time.sleep(GRIPPER_OPERATION_DELAY)  # Precise delay

            

            # Handle movement with FULL validation

            movement_fields = ['x', 'y', 'z', 'r']

            has_movement = any(field in step for field in movement_fields)

            

            if has_movement:

                # Extract coordinates with validation

                try:

                    x = float(step.get('x', 200))  # Default home position

                    y = float(step.get('y', 0))

                    z = float(step.get('z', 150))

                    r = float(step.get('r', 0))

                    

                    step_log += f"Ã°Å¸â€œÂ Moving to ({x:.1f}, {y:.1f}, {z:.1f}, {r:.1f}) "

                    

                    # Execute precise movement

                    success, error = safe_move_to(dobot, x, y, z, r, is_trajectory=True)

                    if success:

                        step_log += "Ã¢Å“â€¦ SUCCESS"

                    else:

                        step_log += f"Ã¢ÂÅ’ FAILED: {error}"

                        # CONTINUE with next step instead of stopping entire trajectory

                        print(step_log)

                        execution_log.append(step_log)

                        print(f"Ã¢Å¡Â Ã¯Â¸Â  Continuing to next step despite failure...")

                        

                except (ValueError, TypeError) as e:

                    step_log += f"Ã¢ÂÅ’ COORDINATE ERROR: {e}"

                    print(step_log)

                    execution_log.append(step_log)

                    break  # Stop on coordinate errors

            

            # Handle pause commands

            if 'pause' in step:

                try:

                    pause_time = float(step['pause'])

                    if pause_time > 0:

                        actual_pause = max(MINIMUM_PAUSE_TIME, pause_time * 0.3)  # Optimize but preserve minimum

                        step_log += f"Ã¢ÂÂ±Ã¯Â¸Â Pausing {actual_pause:.2f}s "

                        time.sleep(actual_pause)

                except (ValueError, TypeError):

                    step_log += "Ã¢ÂÅ’ Invalid pause value "

            

            print(step_log)

            execution_log.append(step_log)

        

        print("Ã¢Å“â€¦ PRECISE Trajectory execution completed")

        print(f"Ã°Å¸â€œÂ Execution Summary: {len(execution_log)} steps processed")

        

    except Exception as e:

        print(f"Ã¢ÂÅ’ Critical error in trajectory execution: {e}")

        import traceback

        traceback.print_exc()

    finally:

        # Restore original speed settings

        try:

            dobot.speed(velocity=original_velocity, acceleration=original_acceleration)

            print(f"Ã°Å¸â€â€ž Restored original speed: {original_velocity}/{original_acceleration}")

        except:

            pass

        is_executing = False

@app.route('/manual', methods=['POST'])

def manual():

    """Manual control"""

    if not dobot:

        return jsonify({'error': 'Dobot not connected'}), 503

    

    data = request.json

    if not data:

        return jsonify({'error': 'No data'}), 400

    

    command = data.get('command')

    params = data.get('params', {})

    

    if command == 'home':

        success, error = safe_move_to(dobot, 200, 0, 150, 0)

        if success:

            return jsonify({'success': True, 'message': 'Moved home'})

        else:

            return jsonify({'error': error}), 500

    

    elif command == 'move':

        x = float(data.get('x', 200))

        y = float(data.get('y', 0))

        z = float(data.get('z', 150))

        r = float(data.get('r', 0))

        

        success, error = safe_move_to(dobot, x, y, z, r)

        if success:

            return jsonify({'success': True, 'message': f'Moved to {x},{y},{z},{r}'})

        else:

            return jsonify({'error': error}), 500

    

    elif command == 'gripper':

        state = data.get('state', 'open')

        dobot.grip(state.lower() == 'close')

        time.sleep(0.5)

        return jsonify({'success': True, 'message': f'Gripper {state}'})

    

    elif command == 'get_position':

        try:

            pose = dobot.pose()

            return jsonify({

                'success': True,

                'position': {

                    'x': round(pose[0], 2),

                    'y': round(pose[1], 2),

                    'z': round(pose[2], 2),

                    'r': round(pose[3], 2)

                }

            })

        except Exception as e:

            return jsonify({'error': str(e)}), 500

    

    elif command == 'pause':

        seconds = float(params.get('seconds', 1))

        time.sleep(seconds)

        return jsonify({'success': True, 'message': f'Paused for {seconds} seconds'})

    

    elif command == 'gripper_rotate':

        angle = float(params.get('angle', 0))

        # For gripper rotation, we'll use the R parameter

        try:

            current_pose = dobot.pose()

            new_r = current_pose[3] + angle

            # Keep within limits

            new_r = max(-180, min(180, new_r))

            success, error = safe_move_to(dobot, current_pose[0], current_pose[1], current_pose[2], new_r)

            if success:

                return jsonify({'success': True, 'message': f'Gripper rotated by {angle} degrees to {new_r} degrees'})

            else:

                return jsonify({'error': error}), 500

        except Exception as e:

            return jsonify({'error': str(e)}), 500

    

    elif command == 'move_relative':

        # Move relative to current position

        dx = float(params.get('dx', 0))

        dy = float(params.get('dy', 0))

        dz = float(params.get('dz', 0))

        dr = float(params.get('dr', 0))

        

        try:

            current_pose = dobot.pose()

            new_x = current_pose[0] + dx

            new_y = current_pose[1] + dy

            new_z = current_pose[2] + dz

            new_r = current_pose[3] + dr

            

            success, error = safe_move_to(dobot, new_x, new_y, new_z, new_r)

            if success:

                return jsonify({'success': True, 'message': f'Moved by ({dx},{dy},{dz},{dr}) to ({new_x:.1f},{new_y:.1f},{new_z:.1f},{new_r:.1f})'})

            else:

                return jsonify({'error': error}), 500

        except Exception as e:

            return jsonify({'error': str(e)}), 500

    

    return jsonify({'error': f'Unknown command: {command}'}), 400
