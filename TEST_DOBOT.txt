#!/usr/bin/env python3
"""
DOBOT HARD RESET - Reset matériel complet
"""

from pydobot import Dobot
import time
import subprocess
import sys

PORT = '/dev/ttyUSB0'

def usb_power_cycle():
    """Coupe et restaure l'alimentation USB"""
    print("🔌 Cycle d'alimentation USB...")
    
    try:
        # Trouver le port USB du Dobot
        result = subprocess.run(['lsusb'], capture_output=True, text=True)
        print(result.stdout)
        
        # Méthode 1: Déconnecter/reconnecter via sysfs
        subprocess.run(['sudo', 'sh', '-c', 'echo 0 > /sys/bus/usb/devices/usb1/authorized'], capture_output=True)
        time.sleep(2)
        subprocess.run(['sudo', 'sh', '-c', 'echo 1 > /sys/bus/usb/devices/usb1/authorized'], capture_output=True)
        time.sleep(3)
        
        print("✅ Cycle USB terminé")
        return True
    except Exception as e:
        print(f"❌ Échec cycle USB: {e}")
        return False

def force_unlock_with_hard_reset():
    """Tentative de déblocage avec reset matériel"""
    
    print("\n" + "="*50)
    print("🔧 DOBOT HARD RESET RECOVERY")
    print("="*50)
    
    # 1. Tenter de lire la position
    try:
        dobot = Dobot(port=PORT, verbose=False)
        pose = dobot.pose()
        print(f"📍 Position avant reset: X={pose[0]:.1f}, Y={pose[1]:.1f}, Z={pose[2]:.1f}")
        dobot.close()
    except:
        print("⚠️ Impossible de lire la position")
    
    # 2. Reset USB matériel
    if usb_power_cycle():
        time.sleep(2)
    
    # 3. Reconnexion
    try:
        dobot = Dobot(port=PORT, verbose=False)
        print("✅ Reconnexion réussie")
        
        # 4. Envoyer un mouvement forcé vers home
        print("📍 Tentative de retour au home...")
        dobot.move_to(200, 0, 150, 0, wait=True)
        time.sleep(1)
        
        # 5. Vérifier la nouvelle position
        pose = dobot.pose()
        print(f"📍 Position après reset: X={pose[0]:.1f}, Y={pose[1]:.1f}, Z={pose[2]:.1f}")
        
        if abs(pose[0] - 200) < 50 and abs(pose[1]) < 50:
            print("🎉 Robot débloqué avec succès!")
            return True
        else:
            print("⚠️ Robot toujours bloqué")
            return False
            
    except Exception as e:
        print(f"❌ Erreur: {e}")
        return False

if __name__ == "__main__":
    if force_unlock_with_hard_reset():
        sys.exit(0)
    else:
        print("\n❌ Le robot est physiquement coincé en butée")
        print("Solution: Déplacer manuellement le bras (UNE SEULE FOIS)")
        sys.exit(1)
